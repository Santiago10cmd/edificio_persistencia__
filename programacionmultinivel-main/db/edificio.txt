    private static String escapar(Object valor) {
        if (valor == null) {
            return "";
        }
        // El & va primero: los demás reemplazos crean textos que empiezan por &
        return valor.toString()
                .replace("&", "&amp;")
                .replace("<", "&lt;")
                .replace(">", "&gt;")
                .replace("\"", "&quot;")
                .replace("'", "&#39;");
    }

    private void crear(Context ctx) {
        Map<String, String> datos = new HashMap<>();
        for (String campo : List.of("nombre", "clase", "nivel", "vida", "ataque", "oro")) {
            String valor = ctx.formParam(campo);
            if (valor != null) {
                datos.put(campo, valor);
            }
        }
        try {
            service.crear(datos.get("nombre"), datos.get("clase"),
                    entero(datos.get("nivel"), "nivel", 1), entero(datos.get("vida"), "vida", 100),
                    entero(datos.get("ataque"), "ataque", 10), dinero(datos.get("oro")));
        } catch (ValidacionException e) {
            ctx.status(400).html(PersonajeVista.formulario(datos, e.getMessage(), motor));
            return;
        }
        ctx.redirect("/personajes");
    }

    public List<Item> agregarItem(int id, String nombreItem, int cantidad) {
        obtener(id);
        if (cantidad < 1 || cantidad > 99) {
            throw new ValidacionException("La cantidad debe estar entre 1 y 99");
        }
        boolean existe = false;
        for (Item item : repositorio.listarCatalogo()) {
            if (item.getNombre().equals(nombreItem)) {
                existe = true;
            }
        }
        if (!existe) {
            throw new ValidacionException("No existe el ítem " + nombreItem);
        }
        repositorio.agregarItem(id, nombreItem, cantidad);
        return repositorio.listarInventario(id);
    }

    @abstractmethod
    def guardar(self, personaje):
        """Guarda un personaje nuevo, le asigna su id y lo devuelve."""

    def buscar_por_id(self, id):
        """Devuelve el personaje con ese id, o None si no existe."""

    @abstractmethod
    def listar_todos(self):
        """Devuelve una lista NUEVA con todos los personajes."""
